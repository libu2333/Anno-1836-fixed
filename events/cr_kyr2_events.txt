namespace = cr_kyr2_events

cr_kyr2_events.1 = {
	type = country_event

	title = cr_kyr2_events.1.t
	desc = cr_kyr2_events.1.d
	flavor = cr_kyr2_events.1.f

	event_image = {
		video = "ep1_cossacks"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 7

	trigger = {
		c:KYR2 ?= this
		has_journal_entry = je_cr_eastern_burut_clans
		NOT = { has_variable = cr_kyr2_baatyr_opening_resolved }
		ruler = { has_template = KYZ_tailak_batyr }
	}

	immediate = {
		set_variable = cr_kyr2_baatyr_opening_resolved
		ruler = { save_scope_as = tailak_baatyr_scope }
	}

	option = {
		name = cr_kyr2_events.1.a
		set_variable = { name = cr_kyr2_baatyr_support value = 1 months = 60 }
		ai_chance = {
			base = 60
		}
	}

	option = {
		name = cr_kyr2_events.1.b
		set_variable = { name = cr_kyr2_clan_rivalry value = -1 months = 36 }
		ai_chance = {
			base = 30
		}
	}

	option = {
		name = cr_kyr2_events.1.c
		trigger = { exists = c:CHI }
		change_relations = {
			country = c:CHI
			value = 15
		}
		set_variable = { name = cr_kyr2_qing_pressure value = -1 months = 24 }
		set_variable = { name = cr_kyr2_clan_rivalry value = -1 months = 24 }
		ai_chance = {
			base = 10
		}
	}
}
