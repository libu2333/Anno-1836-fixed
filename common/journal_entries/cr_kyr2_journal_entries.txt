je_cr_eastern_burut_clans = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	group = je_group_global_international_situations

	scripted_progress_bar = cr_kyr2_clan_cohesion_bar
	scripted_button = cr_kyr2_call_kurultai_button

	possible = {
		has_game_rule = kokand_occupiation
		c:KYR2 ?= this
	}

	should_be_involved = {
		c:KYR2 ?= this
	}

	immediate = {
		c:KYR2 ?= {
			if = {
				limit = {
					NOT = { has_variable = cr_kyr2_baatyr_story_started }
					ruler = { has_template = KYZ_tailak_batyr }
				}
				set_variable = cr_kyr2_baatyr_story_started
				trigger_event = { id = cr_kyr2_events.1 days = 14 popup = yes }
			}
		}
	}

	custom_completion_header = cr_kyr2_clans_complete_header

	complete = {
		custom_tooltip = {
			text = cr_kyr2_clans_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(cr_kyr2_clan_cohesion_bar)" >= 90
			}
		}
	}

	custom_failure_header = cr_kyr2_clans_fail_header

	fail = {
		custom_tooltip = {
			text = cr_kyr2_clans_fail_tt
			scope:journal_entry = {
				"scripted_bar_progress(cr_kyr2_clan_cohesion_bar)" <= 10
			}
		}
	}

	invalid = {
		OR = {
			NOT = { has_game_rule = kokand_occupiation }
			NOT = { exists = c:KYR2 }
		}
	}

	custom_on_completion_header = cr_kyr2_clans_on_complete_header

	on_complete = {
		c:KYR2 ?= {
			add_modifier = {
				name = cr_kyr2_consolidated_tribes
				days = long_modifier_time
			}
			trigger_event = { id = cr_kyr2_events.6 popup = yes }
		}
	}

	custom_on_failure_header = cr_kyr2_clans_on_fail_header

	on_fail = {
		c:KYR2 ?= {
			add_modifier = {
				name = cr_kyr2_fractured_tribes
				days = long_modifier_time
			}
			trigger_event = { id = cr_kyr2_events.7 popup = yes }
		}
	}

	on_monthly_pulse = {
		random_events = {
			90 = 0
			5 = cr_kyr2_events.2
			5 = cr_kyr2_events.3
		}
	}

	weight = 100
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
}

je_cr_eastern_burut_crossroads = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_global_international_situations

	scripted_progress_bar = cr_kyr2_diplomatic_balance_bar
	scripted_button = cr_kyr2_kokand_envoys_button
	scripted_button = cr_kyr2_qing_envoys_button

	possible = {
		has_game_rule = kokand_occupiation
		exists = c:KYR2
		exists = c:KOK
		exists = c:CHI
	}

	should_be_involved = {
		OR = {
			c:KYR2 ?= this
			c:KOK ?= this
			c:CHI ?= this
		}
	}

	custom_completion_header = cr_kyr2_kokand_outcome_header

	complete = {
		custom_tooltip = {
			text = cr_kyr2_kokand_outcome_tt
			scope:journal_entry = {
				"scripted_bar_progress(cr_kyr2_diplomatic_balance_bar)" >= 100
			}
		}
	}

	custom_failure_header = cr_kyr2_qing_outcome_header

	fail = {
		AND = {
			exists = c:CHI
			custom_tooltip = {
				text = cr_kyr2_qing_outcome_tt
				scope:journal_entry = {
					"scripted_bar_progress(cr_kyr2_diplomatic_balance_bar)" <= -100
				}
			}
		}
	}

	invalid = {
		OR = {
			NOT = { has_game_rule = kokand_occupiation }
			NOT = { exists = c:KYR2 }
			NOT = { exists = c:KOK }
			NOT = { exists = c:CHI }
		}
	}

	custom_on_completion_header = cr_kyr2_kokand_on_outcome_header

	on_complete = {
		c:KYR2 ?= {
			remove_diplomatic_pact = {
				country = c:CHI
				type = tributary
			}
			add_modifier = {
				name = cr_kyr2_kokand_patronage
				days = long_modifier_time
			}
		}
		c:KOK ?= {
			create_diplomatic_pact = {
				country = c:KYR2
				type = tributary
			}
		}
		c:KYR2 ?= {
			trigger_event = { id = cr_kyr2_events.8 popup = yes }
		}
	}

	custom_on_failure_header = cr_kyr2_qing_on_outcome_header

	on_fail = {
		c:KYR2 ?= {
			remove_diplomatic_pact = {
				country = c:KOK
				type = tributary
			}
			add_modifier = {
				name = cr_kyr2_qing_suzerainty
				days = long_modifier_time
			}
		}
		c:CHI ?= {
			create_diplomatic_pact = {
				country = c:KYR2
				type = tributary
			}
		}
		c:KYR2 ?= {
			trigger_event = { id = cr_kyr2_events.9 popup = yes }
		}
	}

	on_monthly_pulse = {
		random_events = {
			94 = 0
			3 = cr_kyr2_events.4
			3 = cr_kyr2_events.5
		}
	}

	weight = 100
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
}
